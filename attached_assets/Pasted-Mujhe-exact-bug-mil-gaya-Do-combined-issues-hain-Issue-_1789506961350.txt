winm2O folder ke andar ja kar ye Python script chalao:

bash
cd ~/Downloads/winm2O
bash
python3 -c "
content = open('electron/main.cjs').read()

old1 = '''  const renderWindow = new BrowserWindow({
    show: false,
    width: 1200,
    height: 900,
    backgroundColor: '#ffffff',
    webPreferences: {
      contextIsolation: true,
      sandbox: true,
    },
  })'''

new1 = '''  const renderWindow = new BrowserWindow({
    show: process.platform === 'win32',
    x: process.platform === 'win32' ? -3000 : undefined,
    y: process.platform === 'win32' ? -3000 : undefined,
    width: 1200,
    height: 900,
    backgroundColor: '#ffffff',
    webPreferences: {
      contextIsolation: true,
      sandbox: true,
    },
  })'''

assert content.count(old1) == 1, f'match1 count: {content.count(old1)}'
content = content.replace(old1, new1)

old2 = '''      if (process.platform === 'win32') {
        renderWindow.setContentSize(displayWidth, finalDisplayHeight)
      }
      const outputScreenshot = await withTimeout('''

new2 = '''      if (process.platform === 'win32') {
        renderWindow.setContentSize(displayWidth, finalDisplayHeight)
        await wait(150)
      }
      const outputScreenshot = await withTimeout('''

assert content.count(old2) == 1, f'match2 count: {content.count(old2)}'
content = content.replace(old2, new2)

open('electron/main.cjs', 'w').write(content)
print('OK - both patches applied successfully')
"
